% =============================================================================
% Facilitron-branded LaTeX starter — styled to match the marketing-pages site.
%
% Use when the markdown → pandoc → xelatex path produces ugly tables
% (3+ tables, columns > 4, content that needs page-breaking with repeated
% headers, or fillable form-style grids).
%
% Usage (the tron:md-to-pdf skill does this for you):
%   1. Resolve this skill's dir robustly (do NOT hardcode a version-pinned path;
%      $CLAUDE_SKILL_DIR is not always exported into the agent's bash):
%        name=md-to-pdf
%        SKILL_DIR="${CLAUDE_SKILL_DIR:-${CLAUDE_PLUGIN_ROOT:+$CLAUDE_PLUGIN_ROOT/skills/$name}}"
%        [ -e "$SKILL_DIR/template.tex" ] || SKILL_DIR="$(find ~/.claude/plugins/cache ~/.claude/plugins/marketplaces "$HOME/Library/Application Support/tron-os/tron-releases/versions" -maxdepth 5 -type d -path "*/skills/$name" 2>/dev/null | while read -r d; do [ -e "$d/template.tex" ] && echo "$d"; done | sort -V | tail -1 || true)"
%      then copy this file, substituting @@SKILLDIR@@ with that dir:
%        sed "s|@@SKILLDIR@@|$SKILL_DIR|g" "$SKILL_DIR/template.tex" \
%           > /tmp/facilitron-md-to-pdf/<slug>.tex
%   2. Edit the eyebrow, title, and content body (search for "EDIT:" markers).
%   3. Build:
%        xelatex -interaction=nonstopmode -output-directory=/tmp/facilitron-md-to-pdf \
%                /tmp/facilitron-md-to-pdf/<slug>.tex
%   4. Open the PDF, then upload to ImageKit the same way as the markdown path.
%
% Design mirrors the site (assets/css/tailwind.css + tailwind.config.ts):
%   - Fonts: Inter (body), Archivo (display; ExtraLight for the hero H1 like
%     the site's weight-200 hero), IBM Plex Mono (figures + eyebrow).
%   - Ink: tron-asphalt-900 (#171b2a), NOT pure black. Links/figures: primary-600.
%   - Stat cards: rounded tron-primary-50 callout boxes with a primary-500 bar.
%   - Tables: gray-200 header + white/gray-100 zebra + thin asphalt rules.
% Fonts are vendored in this skill's fonts/ dir; the @@SKILLDIR@@ token below is
% substituted with $CLAUDE_SKILL_DIR on copy, so it builds from /tmp with no
% system font install regardless of where the plugin is installed.
% =============================================================================

\documentclass[11pt]{article}

% --- Page layout -------------------------------------------------------------
\usepackage[margin=1in]{geometry}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.6em}
\linespread{1.12}

% --- Fonts -------------------------------------------------------------------
\usepackage{fontspec}
\def\facfonts{@@SKILLDIR@@/fonts/}

% Body: Inter. Vendored as a variable font (no static weights ship upstream),
% so bold and italic are synthesized.
\setmainfont{Inter-Variable.ttf}[Path=\facfonts, AutoFakeBold=1.8, AutoFakeSlant=0.15]

% Display: Archivo. ExtraLight + tight tracking for the hero H1 (the site's
% frame-h1 is weight 200); Regular/SemiBold/Bold for section headings.
\newfontfamily\archivo[Path=\facfonts, Extension=.ttf,
  UprightFont=Archivo-Regular, BoldFont=Archivo-Bold]{Archivo-Regular}
\newfontfamily\archivosemi[Path=\facfonts, Extension=.ttf]{Archivo-SemiBold}
\newfontfamily\herofont[Path=\facfonts, Extension=.ttf, LetterSpace=-2.0]{Archivo-ExtraLight}

% Mono / figures: IBM Plex Mono. \num{} sets figures in mono; \eyebrowfont is
% the wide-tracked uppercase label above a title.
\newfontfamily\plexmono[Path=\facfonts, Extension=.otf,
  UprightFont=IBMPlexMono-Regular, BoldFont=IBMPlexMono-SemiBold]{IBMPlexMono-Regular}
\newcommand{\num}[1]{{\plexmono #1}}
\newfontfamily\eyebrowfont[Path=\facfonts, Extension=.otf, LetterSpace=8.0]{IBMPlexMono-Medium}

% --- Colors (resolved tron tokens from assets/css/tailwind.css) --------------
\usepackage[table]{xcolor}
\definecolor{asphaltink}{HTML}{171B2A}   % tron-asphalt-900  body + headings
\definecolor{asphaltlead}{HTML}{3E4564}  % tron-asphalt-700  subtitle / lead
\definecolor{asphaltmute}{HTML}{545C84}  % tron-asphalt-600  fine print / captions
\definecolor{primary500}{HTML}{1670E6}   % tron-primary-500  accent bar
\definecolor{primary600}{HTML}{1059FC}   % tron-primary-600  links + figures
\definecolor{primary50}{HTML}{EFF6FF}    % tron-primary-50   card fill
\definecolor{brand700}{HTML}{0068AB}     % tron-brand-700    eyebrow
\definecolor{tablehead}{HTML}{E5E7EB}    % gray-200          table header
\definecolor{zebra}{HTML}{F3F4F6}        % gray-100          zebra row
\definecolor{ruleclr}{HTML}{C2C6DA}      % tron-asphalt-200  table rules

\usepackage[colorlinks=true,linkcolor=primary600,urlcolor=primary600]{hyperref}

% --- Quotes ------------------------------------------------------------------
% IMPORTANT: never author TeX-style ``...'' or `...' quotes here. The backtick
% opening-quote ligature does NOT map in Archivo/Inter and prints a literal `
% glyph. With xelatex+fontspec, the safe options are:
%   1. type real Unicode curly quotes directly:  “ ” ‘ ’  (render from the font)
%   2. type a straight "..." — \MakeOuterQuote below auto-converts it to a
%      proper curly pair, so you never have to think about it.
%   3. \enquote{...} for a nested/explicit quote.
\usepackage{csquotes}
\MakeOuterQuote{"}

% --- Tables, lists, graphics, boxes ------------------------------------------
\usepackage{graphicx}
\usepackage{longtable}   % tables that break across pages with repeated headers
\usepackage{array}       % column types like p{Wcm}
\usepackage{booktabs}    % \toprule \midrule \bottomrule
\usepackage{enumitem}    % checklist spacing control
\usepackage{amssymb}     % \square for outline checkboxes
\usepackage[most]{tcolorbox} % rounded cards

\arrayrulecolor{ruleclr}
\setlength\LTleft{0pt}
\setlength\LTright{\fill}
\renewcommand{\arraystretch}{1.35}

% Logo lookup — keeps \includegraphics{facilitron-logo.png} resolving from /tmp.
\graphicspath{{@@SKILLDIR@@/}}

% --- Helper macros -----------------------------------------------------------
\newcommand{\fchk}{\raisebox{-0.5pt}{\large$\square$}}   % outline checkbox

% Eyebrow label above a title (uppercase, wide-tracked, brand-700).
\newcommand{\eyebrow}[1]{{\eyebrowfont\fontsize{10}{12}\selectfont\color{brand700} #1}\par\vspace{0.5em}}

% h2-style section heading: Archivo bold, asphalt ink.
\newcommand{\sectionhd}[1]{\vspace{0.7em}{\archivo\bfseries\fontsize{17}{19}\selectfont\color{asphaltink} #1}\vspace{0.35em}\par}

% Rounded stat card (the site "callout" card: primary-50 fill, 16px corners,
% 4px left accent bar). Use \stat{big number}{label} inside a tcbraster.
\newtcolorbox{statcard}{
  enhanced, arc=8pt, boxrule=0pt, colback=primary50, colframe=primary50,
  borderline west={3pt}{0pt}{primary500},
  left=14pt, right=12pt, top=11pt, bottom=11pt,
}
\newcommand{\stat}[2]{%
  \begin{statcard}
    {\plexmono\bfseries\fontsize{30}{32}\selectfont\color{primary600} #1}\\[3pt]
    {\archivo\fontsize{11}{14}\selectfont\color{asphaltink} #2}
  \end{statcard}%
}

% Table header cell type (the gray-200 fill comes from \rowcolor{tablehead}).
\newcommand{\thd}[1]{{\archivosemi\color{asphaltink} #1}}
